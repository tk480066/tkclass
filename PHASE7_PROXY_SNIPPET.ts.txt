import { NextResponse, type NextRequest } from "next/server";
import { createClient } from "@supabase/supabase-js";

async function maintenanceRedirect(request: NextRequest) {
  const pathname = request.nextUrl.pathname;
  if (pathname === "/maintenance" || pathname.startsWith("/api/health") || pathname.startsWith("/_next")) {
    return null;
  }

  const url = process.env.NEXT_PUBLIC_SUPABASE_URL;
  const key = process.env.NEXT_PUBLIC_SUPABASE_PUBLISHABLE_KEY;
  if (!url || !key) return null;

  const supabase = createClient(url, key, { auth: { persistSession: false } });
  const { data } = await supabase.rpc("is_maintenance_mode");
  if (data === true && !pathname.startsWith("/admin")) {
    const destination = request.nextUrl.clone();
    destination.pathname = "/maintenance";
    destination.search = "";
    return NextResponse.redirect(destination);
  }
  return null;
}
